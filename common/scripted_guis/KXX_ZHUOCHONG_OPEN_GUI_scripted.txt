scripted_gui = {
    KXX_ZHUOCHONG_OPEN_GUI = {
        context_type = player_context
        window_name = "KXX_ZHUOCHONG_OPEN_GUI"

        visible = {
            has_country_flag = KXXMOUZ_button_open
        }

        effects = {
            button_2_click = {
                log = "Clicked button_2"
                if = {
                    limit = {
                        has_country_flag = KXXMOUZ_MAIN_OPEN
                    }
                    clr_country_flag = KXXMOUZ_MAIN_OPEN
                }
                else = {
                    set_country_flag = KXXMOUZ_MAIN_OPEN
                }
            }
            button_2_shift_click = {
                clr_country_flag = KXXMOUZ_button_open
            }
        }

        triggers = {
            button_2_click_enabled = {
                always = yes
            }
        }
    }
}
