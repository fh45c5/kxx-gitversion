scripted_gui = {
    KXX_CHI_tot_gui_botton = {
        context_type = player_context
        window_name = "KXX_CHI_tot_gui_botton"

        visible = {
            has_global_flag = show_KXX_CHI_tot_gui_botton
        }

        effects = {
            KXX_CHI_tot_botton_click = {
                log = "Clicked KXX_CHI_tot_botton"
                if = {
                    limit = {
                        NOT = {
                            has_country_flag = show_KXX_CHI_tot_gui
                        }
                    }
                    set_country_flag = show_KXX_CHI_tot_gui
                }
                else = {
                    clr_country_flag = show_KXX_CHI_tot_gui
                }
            }
        }

        triggers = {
            KXX_CHI_tot_botton_click_enabled = {
                always = yes
            }
        }
    }
}
