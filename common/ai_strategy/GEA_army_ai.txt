# GEA infantry/armor recruitment targets; weights among these two roles, not guaranteed fielded division counts.
# Existing special-forces/front-control/unit-cap strategies remain active.
# First cancel KX production.txt infantry/armor contributions for GEA, then apply exactly one stage.
# Recheck offset plans if upstream KX production.txt changes.

GEA_offset_default_unit_production = {
	allowed = { tag = GEA }
	enable = { always = yes }
	abort_when_not_enabled = yes

	ai_strategy = {
		type = role_ratio
		id = infantry
		value = -100
	}
}

# KX-style long-term targets: time/industry/research select the stage, not temporary reserves.
GEA_army_ratio_early = {
	allowed = { tag = GEA }
	enable = {
		always = yes
		NOT = { OR = { AND = { date > 1937.12.31 num_of_military_factories > 14 has_tech = motorised_infantry has_tech = tech_recon } AND = { date > 1939.12.31 num_of_military_factories > 24 has_tech = motorised_infantry has_tech = tech_recon OR = { has_tech = basic_medium_tank has_tech = basic_medium_tank_chassis } } AND = { date > 1941.12.31 num_of_military_factories > 39 has_tech = motorised_infantry has_tech = tech_recon OR = { has_tech = basic_medium_tank has_tech = basic_medium_tank_chassis } } AND = { date > 1943.12.31 num_of_military_factories > 49 has_tech = motorised_infantry has_tech = tech_recon OR = { has_tech = basic_medium_tank has_tech = basic_medium_tank_chassis } } } }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = role_ratio id = infantry value = 95 }
	ai_strategy = { type = role_ratio id = armor_role value = 5 }
}

# Heavy TD infantry takes seven points from the existing infantry share; combined infantry/armor stays unchanged.
GEA_heavy_td_infantry_share = {
	allowed = { tag = GEA }
	enable = { num_of_military_factories > 49 has_tech = tech_engineers has_tech = tech_recon has_tech = tech_field_hospital has_tech = tech_logistics_company has_tech = support_weapons has_tech = interwar_antitank OR = { has_tech = basic_medium_tank has_tech = basic_medium_tank_chassis } date > 1941.12.31 }
	abort_when_not_enabled = yes
	ai_strategy = { type = role_ratio id = infantry value = -7 }
	ai_strategy = { type = role_ratio id = GEA_heavy_td_infantry value = 7 }
}

# The custom role must also obey KX division-limit behavior.
GEA_heavy_td_division_limit = {
	allowed = { tag = GEA NOT = { has_game_rule = { rule = unit_limit_options option = NONE } } }
	enable = { has_reached_maximum_divisions = yes NOT = { surrender_progress > 0.05 } }
	abort_when_not_enabled = yes
	ai_strategy = { type = build_army id = GEA_heavy_td_infantry value = -1000 }
}
