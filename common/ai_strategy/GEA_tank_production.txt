# Light flame support: original bootstrap retained.
GEA_flame_tank_bootstrap = {
	allowed = { original_tag = GEA }
	enable = {
		num_of_military_factories > 24
		has_tech = basic_light_tank_chassis
		is_special_project_completed = sp:sp_land_flamethrower_tank
		has_equipment = { light_tank_flame_chassis < 40 }
		has_equipment = { infantry_equipment > 500 }
		has_equipment = { support_equipment > 20 }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = light_tank_flame_chassis
		value = 1
	}
}

# Independent TD reserves: no flame project, flame stockpile, hospital or engineer dependency at the support stage.
GEA_td_support_bootstrap = {
	allowed = { original_tag = GEA }
	enable = {
		num_of_military_factories > 24
		NOT = { AND = { num_of_military_factories > 49 has_tech = tech_engineers has_tech = tech_recon has_tech = tech_field_hospital has_tech = tech_logistics_company has_tech = support_weapons has_tech = interwar_antitank OR = { has_tech = basic_medium_tank has_tech = basic_medium_tank_chassis } date > 1941.12.31 } }
		has_tech = interwar_antitank
		OR = { has_tech = basic_medium_tank has_tech = basic_medium_tank_chassis }
		has_equipment = { medium_tank_destroyer_chassis < 180 }
		has_equipment = { infantry_equipment > 500 }
		has_equipment = { support_equipment > 50 }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = medium_tank_destroyer_chassis
		value = 1
	}
}
