	on_monthly_USA = {
		effect = {
			if = {
				limit = {
					has_global_flag = USA_civil_war_over_flag
					NOT = {
						has_country_flag = USA_reconstruction_is_over
					}
				}
				add_to_variable = { USA_cal_reconstruction_progress = 3 }
				add_to_variable = { USA_wcc_reconstruction_progress = 3 }
				add_to_variable = { USA_tex_reconstruction_progress = 3 }
				add_to_variable = { USA_csa_reconstruction_progress = 3 }
				add_to_variable = { USA_cor_reconstruction_progress = 3 }
				add_to_variable = { USA_Syndicalist_Resistance = -3 }
				add_to_variable = { USA_Minutemen_Resistance = -3 }
				add_to_variable = { USA_Klan_Resistance = -3 }
				add_to_variable = { USA_Black_legion_Resistance = -3 }
				if = {
					limit = {
						owns_state = 796
						owns_state = 931
						owns_state = 932
						owns_state = 935
						owns_state = 933
						owns_state = 934
						owns_state = 357						
					}
					add_to_variable = { USA_nee_reconstruction_progress = 3 }
				}
			}
		}

						owns_state = 935 #罗德岛

						USA_total_reconstruction_factor_temp = 0
					}

						limit = {
							check_variable = {
								var = USA_csa_reconstruction_progress
								value = 100
								compare = greater_than_or_equals
							}
						}
						set_variable = {
							USA_csa_reconstruction_progress = 100
						}
					}

						limit = {
							check_variable = {
								var = USA_cal_reconstruction_progress

								compare = greater_than_or_equals

						}

							}

							USA_nee_reconstruction_progress = 100

					set_temp_variable = { USA_cal_total_reconstruction_progress = USA_cal_reconstruction_progress }
					multiply_temp_variable = { USA_cal_total_reconstruction_progress = 15 }
					add_to_variable = { USA_total_reconstruction_factor_temp = USA_cal_total_reconstruction_progress }

					multiply_temp_variable = { USA_nee_total_reconstruction_progress = 20 }
					add_to_variable = { USA_total_reconstruction_factor_temp = USA_nee_total_reconstruction_progress }

					divide_variable = { USA_total_reconstruction_factor_temp = 100 }
					set_variable = { USA_total_reconstruction_factor = USA_total_reconstruction_factor_temp }

				}
				else = { #未拥有新英格兰
					set_variable = {
						USA_total_reconstruction_factor_temp = 0
					}
					if = {
						limit = {
							check_variable = {
								var = USA_csa_reconstruction_progress
								value = 100
								compare = greater_than_or_equals
							}
						}

							USA_csa_reconstruction_progress = 100

								var = USA_tex_reconstruction_progress
								value = 100
								compare = greater_than_or_equals
							}
						}
						set_variable = {
							USA_tex_reconstruction_progress = 100

					}

						limit = {
							check_variable = {
								var = USA_cor_reconstruction_progress
								value = 100
								compare = greater_than_or_equals
							}
						}
						set_variable = {
							USA_cor_reconstruction_progress = 100
						}
					}
				#计算西海岸
					set_temp_variable = { temp_factor = USA_cal_reconstruction_progress }
					multiply_variable = { 
						var = temp_factor
						value = 25
					}
					add_to_variable = {
						var = USA_total_reconstruction_factor_temp
						value = temp_factor
					}
				#计算西部
					set_temp_variable = { temp_factor = USA_wcc_reconstruction_progress }
					multiply_variable = {
						var = temp_factor
						value = 15
					}
					add_to_variable = {
						var = USA_total_reconstruction_factor_temp
						value = temp_factor
					}
				#计算中部平原
					set_temp_variable = { temp_factor = USA_tex_reconstruction_progress }
					multiply_variable = { 
						var = temp_factor
						value = 15
					}
					add_to_variable = {
						var = USA_total_reconstruction_factor_temp
						value = temp_factor
					}
				#计算钢带区

					multiply_variable = { 
						var = temp_factor
						value = 25
					}
					add_to_variable = {
						var = USA_total_reconstruction_factor_temp
						value = temp_factor
					}
				#计算南部诸州
					set_temp_variable = { temp_factor = USA_cor_reconstruction_progress }
					multiply_variable = { 
						var = temp_factor
						value = 20
					}
					add_to_variable = {
						var = USA_total_reconstruction_factor_temp
						value = temp_factor
					}
				#最终计算
					divide_variable = {
						var = USA_total_reconstruction_factor_temp
						value = 100
					}
					set_variable = {
						USA_total_reconstruction_factor = USA_total_reconstruction_factor_temp
