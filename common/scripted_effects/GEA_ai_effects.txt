# Only controls new division recruitment; existing queues and reinforcement are not cancelled.
GEA_ai_update_training_reserves = {
	if = {
		limit = { NOT = { has_country_flag = GEA_ai_pause_infantry_training } GEA_ai_infantry_training_critical = yes }
		set_country_flag = GEA_ai_pause_infantry_training
	}
	else_if = {
		limit = { has_country_flag = { flag = GEA_ai_pause_infantry_training days > 6 } GEA_ai_infantry_training_healthy = yes }
		clr_country_flag = GEA_ai_pause_infantry_training
	}
	if = {
		limit = { NOT = { has_country_flag = GEA_ai_pause_armor_training } GEA_ai_armor_training_critical = yes }
		set_country_flag = GEA_ai_pause_armor_training
	}
	else_if = {
		limit = { has_country_flag = { flag = GEA_ai_pause_armor_training days > 6 } GEA_ai_armor_training_healthy = yes }
		clr_country_flag = GEA_ai_pause_armor_training
	}
	if = {
		limit = { NOT = { has_country_flag = GEA_ai_pause_heavy_td_training } GEA_ai_heavy_td_training_critical = yes }
		set_country_flag = GEA_ai_pause_heavy_td_training
	}
	else_if = {
		limit = { has_country_flag = { flag = GEA_ai_pause_heavy_td_training days > 6 } GEA_ai_heavy_td_training_healthy = yes }
		clr_country_flag = GEA_ai_pause_heavy_td_training
	}
}
