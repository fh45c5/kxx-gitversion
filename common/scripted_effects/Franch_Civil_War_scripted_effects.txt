FRG_increase_recovery = {
	if = {
		limit = { has_idea = FRG_civil_war_war_aftermath_idea }
		swap_ideas = {
			remove_idea = FRG_civil_war_war_aftermath_idea
			add_idea = FRG_civil_war_war_aftermath_idea2
		}
	}
	else_if = {
		limit = { has_idea = FRG_civil_war_war_aftermath_idea2 }
		swap_ideas = {
			remove_idea = FRG_civil_war_war_aftermath_idea2
			add_idea = FRG_civil_war_war_aftermath_idea3
		}
	}
	else_if = {
		limit = { has_idea = FRG_civil_war_war_aftermath_idea3 }
		swap_ideas = {
			remove_idea = FRG_civil_war_war_aftermath_idea3
			add_idea = FRG_civil_war_war_aftermath_idea4
		}
	}
}

FRG_civil_war_ends = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_flag = FRG_civil_war_over_flag } }
			FRG = {
				if = {
					limit = { 
                        FRA = {
                            exists = no
                        }
                     }
					add_ideas = {
						civilian_economy
						volunteer_only
						FRG_civil_war_war_aftermath_idea
					}
				}
			}
			OCC = {
				if = {
					limit = { 
                        NOT = {
                            has_war_with = FRA
                        }
                    }
					add_ideas = {
						civilian_economy
						volunteer_only
						FRG_civil_war_war_aftermath_idea
					}
				}
			}
			LRE = {
				if = {
					limit = { 
                        NOT = {
                            has_war_with = FRA
                        }
                    }
					add_ideas = {
						civilian_economy
						volunteer_only
						FRG_civil_war_war_aftermath_idea
					}
				}
			}
			BRI = {
				if = {
					limit = { 
                        NOT = {
                            has_war_with = FRA
                        }
                    }
					add_ideas = {
						civilian_economy
						volunteer_only
						FRG_civil_war_war_aftermath_idea
					}
				}
			}
            NRM = {
				if = {
					limit = { 
                        NOT = {
                            has_war_with = FRA
                        }
                    }
					add_ideas = {
						civilian_economy
						volunteer_only
						FRG_civil_war_war_aftermath_idea
					}
				}
			}
            set_global_flag = FRG_civil_war_over_flag
        }
    }
}
