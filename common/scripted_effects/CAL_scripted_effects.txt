			}
		}

		clr_country_flag = CAL_open_pol_gui
		set_country_flag = CAL_open_jap_gui
		gui_dirty_change_cal = yes
	}

		}

			var = CAL_reform_times
			value = 1
		}		
		add_to_variable = {
			var = econ_progress
			value = 5
		}
		add_to_variable = {

			value = -0.03
