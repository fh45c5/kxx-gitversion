#实际移除一座工厂，再向GEA提供同类型工厂
GEA_aog_transfer_one_factory_effect = {
	if = {
		limit = { 
			any_owned_state = { 
				is_controlled_by = PREV 
				industrial_complex > 0 
			} 
		}
		random_owned_state = {
			limit = { 
				is_controlled_by = PREV 
				industrial_complex > 0 
			}
			remove_building = { 
				type = industrial_complex 
				level = 1 
			}
		}
		GEA = {
			if = {
				limit = { any_owned_state = { is_controlled_by = PREV free_building_slots = { building = industrial_complex size = 1 include_locked = yes } } }
				random_owned_state = {
					limit = { is_controlled_by = PREV free_building_slots = { building = industrial_complex size = 1 include_locked = yes } }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
			}
			else = { add_offsite_building = { type = industrial_complex level = 1 } }
		}
	}
	else_if = {
		limit = { any_owned_state = { is_controlled_by = PREV arms_factory > 0 } }
		random_owned_state = {
			limit = { is_controlled_by = PREV arms_factory > 0 }
			remove_building = { type = arms_factory level = 1 }
		}
		GEA = {
			if = {
				limit = { any_owned_state = { is_controlled_by = PREV free_building_slots = { building = arms_factory size = 1 include_locked = yes } } }
				random_owned_state = {
					limit = { is_controlled_by = PREV free_building_slots = { building = arms_factory size = 1 include_locked = yes } }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = arms_factory level = 1 instant_build = yes }
				}
			}
			else = { add_offsite_building = { type = arms_factory level = 1 } }
		}
	}
}
