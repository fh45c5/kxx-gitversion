TEXCOS_click_effect  = {
	if = {
		limit = {
			NOT = {
				has_country_flag = TEXCOS_OPEN
			}
		}
		set_country_flag = TEXCOS_OPEN
		clr_country_flag = FORD_OPEN
		clr_country_flag = USS_OPEN
		clr_country_flag = BOEING_OPEN
		clr_country_flag = DUPONT_OPEN
	}
	
	else_if = {
		limit = {
			has_country_flag = TEXCOS_OPEN
		}
		clr_country_flag = TEXCOS_OPEN
	}
}
