NPP_PATH2_click_effect  = {
	set_country_flag = NPP_the_form_has_finished
	if = {#总统国会
		limit = {
			check_variable = { TEX_NPP_PRESIDENT > 50 }
		}
		add_to_variable = { NPP_political_power_factor = 0.15 }
	}
	else_if = {
		limit = {
			check_variable = { TEX_NPP_CONGRESS > 50  }
		}
		add_to_variable = { NPP_stability_weekly = 0.005 }
	}

	if = {#支持度
		limit = {
			check_variable = { NPP_LEFT1 > 80 }
		}
		add_to_variable = { NPP_industrial_capacity_factory = 0.10 }
	}
	else_if = {
		limit = {
			check_variable = { NPP_LEFT1 > 40 }
		}
		add_to_variable = { NPP_industrial_capacity_factory  = 0.05 }
	}

	if = {#投票热情
		limit = {
			check_variable = { NPP_LEFT2 > 90 }
		}
		add_to_variable = { NPP_party_popularity_stability_factor = 0.20 }
	}
	else_if = {
		limit = {
			check_variable = { NPP_LEFT2 > 40 }
		}
		add_to_variable = { NPP_party_popularity_stability_factor = 0.1 }
	}

	if = {#民主保障程度
		limit = {
			check_variable = { NPP_MIDDLE2 > 75 }
		}
		add_to_variable = { NPP_production_speed_buildings_factor = 0.1 }
	}
	else_if = {
		limit = {
			check_variable = { NPP_MIDDLE2 > 40 }
		}
		add_to_variable = { NPP_production_speed_buildings_factor = 0.05 }
	}

	if = {#满意度
		limit = {
			check_variable = { NPP_RIGHT2 > 80 }
		}
		add_to_variable = { NPP_consumer_goods_factor = -0.20 }
	}
	else_if = {
		limit = {
			check_variable = { NPP_RIGHT2 > 80 }
		}
		add_to_variable = { NPP_consumer_goods_factor = -0.05 }
	}
	else = {
		add_to_variable = { NPP_consumer_goods_factor = 0.1 }
	}
}
