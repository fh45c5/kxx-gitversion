#	disable_party_popularity_PP_gain = yes -- disables popularity PP gain altogether

add_to_ban = {
	if = {
		limit = {
			NOT = {
				is_in_array = { ban_partners = ban_partner_var }
			}
		}
		custom_effect_tooltip = add_to_ban
	}
	if = {
		limit = {
			NOT = {
				is_in_array = { ban_partners = ban_partner_var }
				has_government = var:ban_partner_var
			}
		}
		add_to_array = { ban_partners = ban_partner_var }
		
	}
}

remove_from_ban = {
	if = {
		limit = {
			is_in_array = { ban_partners = ban_partner_var }
		}
		remove_from_array = { ban_partners = ban_partner_var }
	}
	else = {
		custom_effect_tooltip = remove_from_ban
	}
}

end_ban = {
	clear_array = ban_partners
}

		set_temp_variable = { ban_partner_var = token:national_populist }
