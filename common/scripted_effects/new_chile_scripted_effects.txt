CHL_improve_economy_effect = {
	if = {
		limit = {
			has_idea = CHL_economic_disaster
		}
		swap_ideas = {
			remove_idea = CHL_economic_disaster
			add_idea = CHL_economic_disaster1
		}
	}
	else_if = {
		limit = {
			has_idea = CHL_economic_disaster1
		}
		swap_ideas = {
			remove_idea = CHL_economic_disaster1
			add_idea = CHL_economic_disaster2
		}
	}
	else_if = {
		limit = {
			has_idea = CHL_economic_disaster2
		}
		swap_ideas = {
			remove_idea = CHL_economic_disaster2
			add_idea = CHL_economic_disaster3
		}
	}
	else_if = {
		limit = {
			has_idea = CHL_economic_disaster3
		}
		remove_ideas = CHL_economic_disaster3
	}
	
}
