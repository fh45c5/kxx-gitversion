GEA_szc_relieve_famine_medium_effect = {
	if = {
		limit = {
			has_idea = SZC_famine_idea
			has_dynamic_modifier = {
				modifier = SZC_famine_modifier
			}
			has_variable = SZC_famine_monthly_population
		}
		custom_effect_tooltip = SZC_reduce_famine_moderate_tooltip
		hidden_effect = {
			add_to_variable = {
				SZC_famine_monthly_population = 0.1
			}
			add_to_variable = {
				SZC_famine_stability_factor = 0.01
			}
			add_to_variable = {
				SZC_famine_war_support_factor = 0.04
			}
			subtract_from_variable = {
				SZC_famine_consumer_goods_factor = 0.01
			}
			force_update_dynamic_modifier = yes
			if = {
				limit = {
					check_variable = {
						SZC_famine_monthly_population > -0.01
					}
				}
				SZC_end_famine = yes
			}
		}
	}
}
